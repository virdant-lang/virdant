Package := DocBang? PackageStmt*

PackageStmt :=
      PackageImport
    | Item

PackageImport := "import" Ident

DocString := ("DocComment")+

DocBang := ("DocBang")+

Annotations := Annotation*

Annotation :=
      "@" Ident "(" ArgList? ")"
    | "@" Ident

Item :=
      ModDef
    | StructDef
    | UnionDef
    | EnumDef
    | BuiltinDef
    | SocketDef
    | PlatformDef

ModDef := DocString? Annotations "ext"? "export"? "mod" Ident ForClause? Generics? "{" ModDefStmt* "}"

ForClause := "for" Ofness

StructDef := DocString? Annotations "struct" "type" Ident Generics? "{" StructDefStmt* "}"

PlatformDef := DocString? Annotations "platform" Ident "{" PlatformStmt* "}"

PlatformStmt := ModDefStmtComponent

StructDefStmt := DocString? Annotations Ident ":" Type

UnionDef := DocString? Annotations "union" "type" Ident Generics? "{" UnionDefStmt* "}"

UnionDefStmt := DocString? Annotations Ident ParamList?

ParamList := "(" Params? ")"

Params :=
      Param
    | Params "," Param

Param := DocString? Annotations Ident ":" Type

GenericsParams := "[" GenericsArgList "]"

GenericsArgList := (GenericsArg ",")* GenericsArg

GenericsArg :=
      Nat
    | Type

Generics := "[" KindParamList "]"

KindParamList := (KindParam ",")* KindParam

KindParam := Ident ":" Kind

EnumDef := DocString? Annotations "enum" "type" Ident Generics? "{" EnumDefStmt* "}"

EnumDefStmt :=
      DocString? Annotations Ident "=" Expr
    | DocString? Annotations Ident ParamList?
    | DocString? Annotations Ident ParamList "=" Expr

BuiltinDef := DocString? Annotations "builtin" "type" Ident Generics? "{" "}"

SocketDef := DocString? Annotations "socket" Ident Generics? "{" SocketDefStmt* "}"

SocketDefStmt :=
      DocString? Annotations "cosi" Ident ":" Type
    | DocString? Annotations "soci" Ident ":" Type

ModDefStmtCore :=
      ModDefStmtComponent
    | ModDefStmtDriver
    | ModDefStmtInstance
    | ModDefStmtSocket
    | ModDefStmtUnused
    | ModDefStmtInitial
    | ModDefStmtDependsOn

ModDefStmt :=
      ModDefStmtCore
    | When
    | Match

ModDefStmtComponent :=
      DocString? Annotations "incoming" Ident ":" Type OnClause? ItBlock?
    | DocString? Annotations "outgoing" Ident ":" Type DefaultClause? OnClause? ItBlock?
    | DocString? Annotations "bus" Ident ":" Type ItBlock?
    | DocString? Annotations "net" Ident ":" Type ItBlock?
    | DocString? Annotations "outgoing" "wire" Ident ":" Type DefaultClause? OnClause? ItBlock?
    | DocString? Annotations "outgoing" "reg" Ident ":" Type DefaultClause? OnClause? ItBlock?
    | DocString? Annotations "wire" Ident ":" Type OnClause? ItBlock?
    | DocString? Annotations "reg" Ident ":" Type OnClause? ItBlock?

OnClause := "on" Expr

DefaultClause := "default" Expr

ModDefStmtDriver :=
      Path ":=" Expr
    | Path "<=" Expr
    | Path "#=" Expr
    | Path "#-" Expr
    | Path "#=#" Path
    | Path ":=:" Path

ModDefStmtInstance := DocString? Annotations "mod" Ident "of" Ofness GenericsParams? ItBlock?

ItBlock := ModDefStmtBlock

ModDefStmtSocket :=
      DocString? Annotations "client" "socket" Ident "of" Ofness GenericsParams? ItBlock?
    | DocString? Annotations "server" "socket" Ident "of" Ofness GenericsParams? ItBlock?

When := "when" "{" WhenArm* "}"

WhenArm :=
      "case" Expr "=>" Expr
    | "case" Expr ArmBlock
    | "case" Expr When
    | "case" Expr Match
    | "else" "=>" Expr
    | "else" ArmBlock
    | "else" When
    | "else" Match

Match := "match" Expr "{" MatchArm* "}"

MatchArm :=
      "case" Pat "=>" Expr
    | "case" Pat ArmBlock
    | "case" Pat When
    | "case" Pat Match
    | "else" "=>" Expr
    | "else" ArmBlock
    | "else" When
    | "else" Match

ModDefStmtUnused := "unused" Path

ModDefStmtInitial := "initial" Path "<=" Expr

ModDefStmtDependsOn := Path "dependson" Path

ModDefStmtBlock := "{" ModDefStmt* "}"

Kind := Ident

Type := Ofness GenericsParams?

Expr :=
      When
    | Match
    | ExprStruct
    | ExprBinOpLogical

ArmBlock :=
      "{" ModDefStmt* "}"
    | "{" ArmBlockExpr "}"

ArmBlockExpr :=
      ArmBlockExpr BinOpLogical ArmBlockCompare
    | ArmBlockCompare

ArmBlockCompare :=
      ArmBlockCompare BinOpCompareNoLe ExprBinOpAdditive
    | ExprBinOpAdditive

BinOpCompareNoLe :=
      "<"
    | ">"
    | ">="
    | "=="
    | "!="

Pat :=
      "#" Ident
    | "@" Ident
    | "@" Ident "(" PatList ")"
    | WordLit
    | BitLit
    | Ident
    | "dontcare"

PatList :=
      (Pat ",")+ Pat ","?
    | Pat ","?
    |

ExprStruct := "$" "{" AssignList "}"

AssignList :=
      (Assign ",")* Assign ","?
    | Expr ","?
    |

Assign := Ident "=" Expr

BinOpLogical :=
      "&&"
    | "||"
    | "^^"

ExprBinOpLogical :=
      ExprBinOpLogical BinOpLogical ExprBinOpCompare
    | ExprBinOpCompare

BinOpCompare :=
      "<"
    | "<="
    | ">"
    | ">="
    | "=="
    | "!="

ExprBinOpCompare :=
      ExprBinOpCompare BinOpCompare ExprBinOpAdditive
    | ExprBinOpAdditive

BinOpAdditive :=
      "+"
    | "-"
    | "&"
    | "|"
    | "^"

ExprBinOpAdditive :=
      ExprBinOpAdditive BinOpAdditive ExprUnOp
    | ExprUnOp

UnOp :=
      "-"
    | "~"
    | "!"

ExprUnOp :=
      UnOp ExprUnOp
    | "tap" Path
    | ExprAscription

ExprAscription :=
      ExprPrimary ":" Type
    | ExprPrimary

ExprPrimary :=
      Ofness "(" ArgList? ")"
    | ExprPrimary "->" Ident
    | ExprPrimary "[" "dyn" Expr "]"
    | ExprPrimary "[" Index "]"
    | ExprPrimary "[" Index ".." Index "]"
    | "@" Ident "(" ArgList? ")"
    | ExprAtom

ExprAtom :=
      Path
    | WordLit
    | BitLit
    | "Str"
    | "#" Ident
    | "@" Ident
    | "?"
    | "dontcare"
    | "(" Expr ")"

ArgList :=
      (Expr ",")+ Expr ","?
    | Expr ","?

Ofness :=
      Ident
    | Ident "::" Ident

Path :=
      Ident ("." Ident)*
    | "it" ("." Ident)*

Index := "Nat"

Width := "Nat"

WordLit :=
      "Nat"
    | "Word"

BitLit :=
      "true"
    | "false"

Ident := "Ident"

Nat := "Nat"

Str := "Str"
